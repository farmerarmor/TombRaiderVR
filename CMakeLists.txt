cmake_minimum_required(VERSION 3.24)
project(TombRaiderVR VERSION 0.1.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_API_LAYERS OFF CACHE BOOL "" FORCE)
set(BUILD_CONFORMANCE_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_ALL_EXTENSIONS OFF CACHE BOOL "" FORCE)
set(BUILD_LOADER ON CACHE BOOL "" FORCE)
set(DYNAMIC_LOADER OFF CACHE BOOL "" FORCE)
FetchContent_Declare(openxr GIT_REPOSITORY https://github.com/KhronosGroup/OpenXR-SDK-Source.git GIT_TAG c07ad64839653712190e05dbd8cf460e1d239513 GIT_SUBMODULES "")
FetchContent_Declare(minhook GIT_REPOSITORY https://github.com/TsudaKageyu/minhook.git GIT_TAG c3fcafdc10146beb5919319d0683e44e3c30d537 GIT_SUBMODULES "")
FetchContent_MakeAvailable(openxr minhook)
add_library(NativeBridge STATIC src/NativeBridge.cpp)
target_include_directories(NativeBridge PUBLIC src)
target_compile_definitions(NativeBridge PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX XR_USE_PLATFORM_WIN32 XR_USE_GRAPHICS_API_D3D11)
target_link_libraries(NativeBridge PUBLIC openxr_loader d3d11 dxgi d3dcompiler)
add_executable(TombRaiderVRProbe tools/Probe.cpp)
add_executable(TombRaiderVRCameraMathProbe tools/CameraMathProbe.cpp)
target_include_directories(TombRaiderVRCameraMathProbe PRIVATE src)
target_compile_definitions(TombRaiderVRCameraMathProbe PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(TombRaiderVRCameraMathProbe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_link_libraries(TombRaiderVRProbe PRIVATE NativeBridge)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  add_executable(TombRaiderVRHost src/Host.cpp)
  target_link_libraries(TombRaiderVRHost PRIVATE NativeBridge)
  set_target_properties(TombRaiderVRHost PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  set_target_properties(TombRaiderVRProbe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
  return()
endif()
add_library(atidxx32 SHARED vendor/AmdQbProxy/AmdQbProxy.cpp vendor/AmdQbProxy/AmdQbProxy.def)
add_executable(TombRaiderVRStereoSizingProbe tools/StereoSizingProbe.cpp)
target_link_libraries(TombRaiderVRStereoSizingProbe PRIVATE d3d11 dxgi user32)
set_target_properties(TombRaiderVRStereoSizingProbe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_library(NativeTransport STATIC src/NativeTransport.cpp src/EngineCamera.cpp src/EngineDisplay.cpp )
target_include_directories(NativeTransport PUBLIC src)
target_compile_definitions(NativeTransport PUBLIC WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(NativeTransport PUBLIC d3d11 dxgi minhook)
add_executable(TombRaiderVRTransportProbe tools/TransportProbe.cpp)
target_link_libraries(TombRaiderVRTransportProbe PRIVATE NativeTransport)
set_target_properties(TombRaiderVRTransportProbe PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_link_libraries(atidxx32 PRIVATE NativeTransport minhook user32 gdi32)
add_library(d3d11_proxy SHARED vendor/D3d11VendorProxy/D3d11VendorProxy.cpp vendor/D3d11VendorProxy/D3d11VendorProxy.def)
target_link_libraries(d3d11_proxy PRIVATE minhook user32 advapi32)
target_compile_definitions(d3d11_proxy PRIVATE BLOCK_NVAPI_STEREO)
set_target_properties(d3d11_proxy PROPERTIES OUTPUT_NAME d3d11)
add_library(atiadlxy SHARED vendor/AmdAdlProxy/AmdAdlProxy.cpp vendor/AmdAdlProxy/AmdAdlProxy.def)
target_link_libraries(atiadlxy PRIVATE user32)
foreach(t NativeBridge NativeTransport atidxx32 d3d11_proxy atiadlxy TombRaiderVRProbe)
  target_compile_definitions(${t} PRIVATE UNICODE _UNICODE)
  target_compile_options(${t} PRIVATE /W3 /EHsc /utf-8)
  set_target_properties(${t} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
endforeach()
